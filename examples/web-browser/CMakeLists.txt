cmake_minimum_required(VERSION 3.25)
project(crtweb_browser CXX)
if(NOT CMAKE_SYSROOT)
  message(FATAL_ERROR "use the CRT SDK's crt-toolchain.cmake")
endif()
if(NOT TARGET crtui_skia)
  file(READ "${CMAKE_SYSROOT}/manifest.json" manifest)
  string(JSON CRT_STAGE_TARGET_OS GET "${manifest}" target os)
  string(JSON CRT_STAGE_TARGET_ARCH GET "${manifest}" target arch)
  include("${CMAKE_SYSROOT}/share/crt/cmake/crt_upper_imports.cmake")
endif()
add_executable(crtweb_browser main.cc)
if(CRT_STAGE_TARGET_OS STREQUAL "linux")
  set_target_properties(crtweb_browser PROPERTIES
    BUILD_RPATH "$ORIGIN/../../lib;${CMAKE_SYSROOT}/lib" INSTALL_RPATH "$ORIGIN/../../lib")
elseif(CRT_STAGE_TARGET_OS STREQUAL "macos")
  set_target_properties(crtweb_browser PROPERTIES
    BUILD_RPATH "@loader_path/../../lib;${CMAKE_SYSROOT}/lib" INSTALL_RPATH "@loader_path/../../lib")
endif()
target_compile_features(crtweb_browser PRIVATE cxx_std_17)
target_include_directories(crtweb_browser PRIVATE "${CMAKE_SYSROOT}/include/crtweb")
if(CRT_STAGE_SOURCE_ROOT)
  target_include_directories(crtweb_browser PRIVATE
    "${CRT_STAGE_SOURCE_ROOT}/libcrtweb/include" "${CRT_STAGE_SOURCE_ROOT}/libcrtweb/platform")
endif()
target_link_libraries(crtweb_browser PRIVATE crtui_skia crtgfx_skia)
install(TARGETS crtweb_browser RUNTIME DESTINATION examples/bin)
